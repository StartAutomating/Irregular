# Matches Output from git diff
(?m)(?<Git_DiffHeader>
(?m)^diff                                          # diff line start
\s--git\s                                          # whitespace, then --git, then whitespace
(?<From>\S+)                                       # From
\s                                                 # Whitespace
(?<To>\S+)                                         # To
(?<ExtendedHeader>(?:.|\s){0,}?(?=\z|^index\s{1})) # Until 'index'

)
(?<Git_DiffRanges>(?:.|\s){0,}?(?=\z|^diff))
